# Stage 1: Build the Go binary
FROM golang:1.22-alpine AS builder

WORKDIR /app
COPY go.mod ./
COPY main.go ./

RUN CGO_ENABLED=0 GOOS=linux go build -o /call-events .

# Stage 2: Minimal non-root runtime image
FROM alpine:3.20

RUN addgroup -S appgroup && adduser -S appuser -G appgroup

WORKDIR /app
COPY --from=builder /call-events /app/call-events

EXPOSE 8080
USER 10001:10001

ENTRYPOINT ["/app/call-events"]
